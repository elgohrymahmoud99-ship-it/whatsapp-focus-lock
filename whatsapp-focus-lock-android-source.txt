package com.focuslock.whatsapp

import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import java.util.concurrent.atomic.AtomicBoolean
import java.util.concurrent.atomic.AtomicLong

/**
 * Thread-safe singleton holding the global state of the focus lock.
 * Designed for concurrent access across Background Services (NotificationListener,
 * AccessibilityService) and Jetpack Compose UI.
 */
object AppLockState {

    // Atomic primitives for high-performance non-blocking reads in AccessibilityService
    private val _isUnlockedDueToNotification = AtomicBoolean(false)
    private val _isLockActive = AtomicBoolean(true)
    private val _lastNotificationTimestamp = AtomicLong(0L)
    private val _emergencyBypassUntilTimestamp = AtomicLong(0L)

    // Reactive StateFlows for Jetpack Compose UI Observation
    private val _isUnlockedFlow = MutableStateFlow(false)
    val isUnlockedDueToNotificationFlow: StateFlow<Boolean> = _isUnlockedFlow.asStateFlow()

    private val _isLockActiveFlow = MutableStateFlow(true)
    val isLockActiveFlow: StateFlow<Boolean> = _isLockActiveFlow.asStateFlow()

    private val _lastTimestampFlow = MutableStateFlow(0L)
    val lastNotificationTimestampFlow: StateFlow<Long> = _lastTimestampFlow.asStateFlow()

    private val _emergencyBypassUntilFlow = MutableStateFlow(0L)
    val emergencyBypassUntilFlow: StateFlow<Long> = _emergencyBypassUntilFlow.asStateFlow()

    /**
     * Timestamp (Epoch ms) until which Quick Emergency Bypass remains active.
     */
    var emergencyBypassUntilTimestamp: Long
        get() = _emergencyBypassUntilTimestamp.get()
        @Synchronized
        set(value) {
            _emergencyBypassUntilTimestamp.set(value)
            _emergencyBypassUntilFlow.value = value
        }

    /**
     * Returns true if emergency bypass is currently active and within the time window.
     */
    val isEmergencyBypassActive: Boolean
        get() {
            val until = _emergencyBypassUntilTimestamp.get()
            return until > System.currentTimeMillis()
        }

    /**
     * Activates a 5-minute (default) emergency bypass window.
     */
    @Synchronized
    fun activateEmergencyBypass(durationMillis: Long = 5 * 60 * 1000L) {
        emergencyBypassUntilTimestamp = System.currentTimeMillis() + durationMillis
    }

    /**
     * Immediately cancels the emergency bypass.
     */
    @Synchronized
    fun clearEmergencyBypass() {
        emergencyBypassUntilTimestamp = 0L
    }

    /**
     * Flag indicating whether WhatsApp is temporarily permitted because an active
     * unread notification was posted and has not yet been dismissed or closed.
     */
    var isUnlockedDueToNotification: Boolean
        get() = _isUnlockedDueToNotification.get()
        @Synchronized
        set(value) {
            _isUnlockedDueToNotification.set(value)
            _isUnlockedFlow.value = value
        }

    /**
     * Master user switch enabling or disabling the targeted focus lock.
     */
    var isLockActive: Boolean
        get() = _isLockActive.get()
        @Synchronized
        set(value) {
            _isLockActive.set(value)
            _isLockActiveFlow.value = value
        }

    /**
     * Timestamp (Epoch ms) of the most recent notification received from WhatsApp.
     */
    var lastNotificationTimestamp: Long
        get() = _lastNotificationTimestamp.get()
        @Synchronized
        set(value) {
            _lastNotificationTimestamp.set(value)
            _lastTimestampFlow.value = value
        }

    /**
     * Convenience helper to reset session state upon leaving WhatsApp.
     */
    @Synchronized
    fun resetUnlockState() {
        isUnlockedDueToNotification = false
    }
}

package com.focuslock.whatsapp

import android.service.notification.NotificationListenerService
import android.service.notification.StatusBarNotification
import android.util.Log

/**
 * Service that listens for incoming and dismissed notifications.
 * When a notification from "com.whatsapp" is posted, it allows temporary access.
 * When all notifications from "com.whatsapp" are dismissed, access is re-locked.
 */
class WhatsAppNotificationListener : NotificationListenerService() {

    companion object {
        private const val TAG = "WhatsAppNotifListener"
        const val WHATSAPP_PACKAGE = "com.whatsapp"
    }

    override fun onListenerConnected() {
        super.onListenerConnected()
        Log.i(TAG, "NotificationListenerService connected. Checking existing active notifications.")
        checkRemainingWhatsAppNotifications()
    }

    override fun onNotificationPosted(sbn: StatusBarNotification?) {
        super.onNotificationPosted(sbn)
        if (sbn == null) return

        val pkgName = sbn.packageName
        if (pkgName == WHATSAPP_PACKAGE) {
            val postTime = sbn.postTime
            Log.d(TAG, "WhatsApp notification posted at $postTime. Unlocking access.")
            AppLockState.isUnlockedDueToNotification = true
            AppLockState.lastNotificationTimestamp = postTime
        }
    }

    override fun onNotificationRemoved(sbn: StatusBarNotification?) {
        super.onNotificationRemoved(sbn)
        if (sbn == null) return

        val pkgName = sbn.packageName
        if (pkgName == WHATSAPP_PACKAGE) {
            Log.d(TAG, "WhatsApp notification removed. Checking if other WhatsApp notifications remain.")
            checkRemainingWhatsAppNotifications()
        }
    }

    /**
     * Inspects active notifications in the status bar. If there are no longer
     * any notifications from com.whatsapp, locks access.
     */
    private fun checkRemainingWhatsAppNotifications() {
        try {
            val activeNotifications: Array<StatusBarNotification>? = activeNotifications
            val hasWhatsAppNotification = activeNotifications?.any { it.packageName == WHATSAPP_PACKAGE } ?: false

            if (!hasWhatsAppNotification) {
                Log.d(TAG, "No active WhatsApp notifications in status bar. Re-locking access.")
                AppLockState.isUnlockedDueToNotification = false
            } else {
                Log.d(TAG, "Active WhatsApp notifications still present.")
                AppLockState.isUnlockedDueToNotification = true
            }
        } catch (e: SecurityException) {
            Log.e(TAG, "SecurityException while reading active notifications: ${e.message}", e)
        }
    }
}

package com.focuslock.whatsapp

import android.accessibilityservice.AccessibilityService
import android.content.Context
import android.graphics.PixelFormat

/**
 * Foreground Application Monitor Service using Android AccessibilityService.
 * Observes TYPE_WINDOW_STATE_CHANGED events to detect whenever WhatsApp
 * is brought to the foreground or exited.
 */
class WhatsAppBlockerService : AccessibilityService() {

    companion object {
        private const val TAG = "WhatsAppBlockerService"
        const val WHATSAPP_PACKAGE = "com.whatsapp"
    }

    private var currentForegroundPackage: String? = null
    private var windowManager: WindowManager? = null
    private var overlayView: View? = null
    private val mainHandler = Handler(Looper.getMainLooper())

    override fun onServiceConnected() {
        super.onServiceConnected()
        Log.i(TAG, "WhatsAppBlockerService connected.")
        windowManager = getSystemService(Context.WINDOW_SERVICE) as? WindowManager
    }

    private var countdownOverlayView: View? = null
    private var countdownRunnable: Runnable? = null

    override fun onAccessibilityEvent(event: AccessibilityEvent?) {
        if (event == null || event.eventType != AccessibilityEvent.TYPE_WINDOW_STATE_CHANGED) {
            return
        }

        val packageNameCharSequence = event.packageName ?: return
        val newPackage = packageNameCharSequence.toString()

        // Ignore system UI overlays (keyguard, volume panel, notifications shade)
        if (newPackage == "android" || newPackage == "com.android.systemui") {
            return
        }

        val previousPackage = currentForegroundPackage
        currentForegroundPackage = newPackage

        Log.d(TAG, "Foreground changed: previous=[$previousPackage] -> current=[$newPackage]")

        // Case 1: User transitioned OUT of WhatsApp to any other app
        if (previousPackage == WHATSAPP_PACKAGE && newPackage != WHATSAPP_PACKAGE) {
            Log.i(TAG, "User exited WhatsApp to $newPackage. Resetting unlock notification state.")
            AppLockState.resetUnlockState()
            removeOverlay()
            removeCountdownOverlay()
        }

        // Case 2: Target app is WhatsApp
        if (newPackage == WHATSAPP_PACKAGE) {
            handleWhatsAppForeground()
        } else {
            removeOverlay()
            removeCountdownOverlay()
        }
    }

    private fun handleWhatsAppForeground() {
        val isLocked = AppLockState.isLockActive
        val isUnlocked = AppLockState.isUnlockedDueToNotification || AppLockState.isEmergencyBypassActive

        Log.d(TAG, "WhatsApp opened. isLockActive=$isLocked, isUnlocked=$isUnlocked (notif=${AppLockState.isUnlockedDueToNotification}, bypass=${AppLockState.isEmergencyBypassActive})")

        if (isLocked && !isUnlocked) {
            Log.w(TAG, "BLOCKED: WhatsApp accessed without notification or active bypass! Enforcing focus lock.")
            removeCountdownOverlay()

            // Method 1: Immediate minimize via GLOBAL_ACTION_HOME
            val homeInvoked = performGlobalAction(GLOBAL_ACTION_HOME)
            Log.d(TAG, "performGlobalAction(GLOBAL_ACTION_HOME) status: $homeInvoked")

            // Method 2: Present full-screen blocking overlay if SYSTEM_ALERT_WINDOW is granted
            if (Settings.canDrawOverlays(this)) {
                mainHandler.post {
                    showBlockingOverlay()
                }
            }
        } else {
            Log.i(TAG, "ALLOWED: WhatsApp opened with active authorization.")
            removeOverlay()

            // If unlocked specifically due to Emergency Bypass, show persistent on-screen countdown timer
            if (AppLockState.isEmergencyBypassActive && Settings.canDrawOverlays(this)) {
                mainHandler.post {
                    showCountdownOverlay()
                }
            } else {
                removeCountdownOverlay()
            }
        }
    }

    /**
     * Renders a system-level blocking overlay window on top of WhatsApp.
     */
    private fun showBlockingOverlay() {
        if (overlayView != null) return // Already displayed

        val wm = windowManager ?: return

        val layoutType = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY
        } else {
            @Suppress("DEPRECATION")
            WindowManager.LayoutParams.TYPE_PHONE
        }

        val params = WindowManager.LayoutParams(
            WindowManager.LayoutParams.MATCH_PARENT,
            WindowManager.LayoutParams.MATCH_PARENT,
            layoutType,
            WindowManager.LayoutParams.FLAG_NOT_TOUCH_MODAL or
                    WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN or
                    WindowManager.LayoutParams.FLAG_FULLSCREEN,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.CENTER
        }

        val layout = createOverlayView()
        overlayView = layout

        try {
            wm.addView(layout, params)
            Log.i(TAG, "Blocking overlay added successfully.")
        } catch (e: Exception) {
            Log.e(TAG, "Failed to add blocking overlay: ${e.message}", e)
        }
    }

    private fun createOverlayView(): View {
        val container = android.widget.LinearLayout(this).apply {
            orientation = android.widget.LinearLayout.VERTICAL
            gravity = Gravity.CENTER
            setBackgroundColor(android.graphics.Color.argb(245, 18, 24, 38))
            setPadding(64, 64, 64, 64)
        }

        val icon = TextView(this).apply {
            text = "🔒"
            textSize = 48f
            gravity = Gravity.CENTER
            setPadding(0, 0, 0, 24)
        }

        val title = TextView(this).apply {
            text = "WhatsApp is Locked"
            textSize = 24f
            setTextColor(android.graphics.Color.WHITE)
            typeface = android.graphics.Typeface.DEFAULT_BOLD
            gravity = Gravity.CENTER
            setPadding(0, 0, 0, 16)
        }

        val message = TextView(this).apply {
            text = "WhatsApp is locked in Focus Mode.\nIt only unlocks when a new WhatsApp notification arrives."
            textSize = 15f
            setTextColor(android.graphics.Color.argb(200, 255, 255, 255))
            gravity = Gravity.CENTER
            setPadding(0, 0, 0, 36)
        }

        // Quick Emergency Bypass Button (5 Minutes)
        val bypassButton = Button(this).apply {
            text = "⚡ Quick Emergency Bypass (5 Min)"
            setBackgroundColor(android.graphics.Color.rgb(245, 158, 11))
            setTextColor(android.graphics.Color.BLACK)
            typeface = android.graphics.Typeface.DEFAULT_BOLD
            setOnClickListener {
                Log.i(TAG, "User activated Quick Emergency Bypass for 5 minutes.")
                AppLockState.activateEmergencyBypass(5 * 60 * 1000L)
                removeOverlay()
                showCountdownOverlay()
            }
        }

        val spacer = View(this).apply {
            layoutParams = android.widget.LinearLayout.LayoutParams(
                android.widget.LinearLayout.LayoutParams.MATCH_PARENT,
                24
            )
        }

        val exitButton = Button(this).apply {
            text = "Return to Home"
            setBackgroundColor(android.graphics.Color.rgb(37, 211, 102))
            setTextColor(android.graphics.Color.BLACK)
            setOnClickListener {
                performGlobalAction(GLOBAL_ACTION_HOME)
                removeOverlay()
            }
        }

        container.addView(icon)
        container.addView(title)
        container.addView(message)
        container.addView(bypassButton)
        container.addView(spacer)
        container.addView(exitButton)

        return container
    }

    /**
     * Renders an unobtrusive floating pill on screen with a live 5-minute countdown.
     */
    private fun showCountdownOverlay() {
        if (countdownOverlayView != null) return

        val wm = windowManager ?: return

        val layoutType = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY
        } else {
            @Suppress("DEPRECATION")
            WindowManager.LayoutParams.TYPE_PHONE
        }

        val params = WindowManager.LayoutParams(
            WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.WRAP_CONTENT,
            layoutType,
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
                    WindowManager.LayoutParams.FLAG_NOT_TOUCH_MODAL,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.TOP or Gravity.CENTER_HORIZONTAL
            y = 120
        }

        val pillLayout = android.widget.LinearLayout(this).apply {
            orientation = android.widget.LinearLayout.HORIZONTAL
            gravity = Gravity.CENTER_VERTICAL
            setBackgroundColor(android.graphics.Color.argb(230, 20, 28, 38))
            setPadding(36, 16, 36, 16)
        }

        val timerText = TextView(this).apply {
            setTextColor(android.graphics.Color.rgb(251, 191, 36))
            textSize = 14f
            typeface = android.graphics.Typeface.MONOSPACE
            text = "⚡ Bypass: 05:00"
            setPadding(0, 0, 24, 0)
        }

        val lockNowBtn = TextView(this).apply {
            setTextColor(android.graphics.Color.rgb(248, 113, 113))
            textSize = 12f
            text = "✕ Lock Now"
            setOnClickListener {
                AppLockState.clearEmergencyBypass()
                removeCountdownOverlay()
                handleWhatsAppForeground()
            }
        }

        pillLayout.addView(timerText)
        pillLayout.addView(lockNowBtn)
        countdownOverlayView = pillLayout

        try {
            wm.addView(pillLayout, params)

            countdownRunnable = object : Runnable {
                override fun run() {
                    val remainingMs = AppLockState.emergencyBypassUntilTimestamp - System.currentTimeMillis()
                    if (remainingMs > 0) {
                        val minutes = (remainingMs / 1000) / 60
                        val seconds = (remainingMs / 1000) % 60
                        timerText.text = String.format("⚡ Bypass: %02d:%02d", minutes, seconds)
                        mainHandler.postDelayed(this, 1000)
                    } else {
                        Log.i(TAG, "Emergency Bypass expired. Re-evaluating focus lock.")
                        AppLockState.clearEmergencyBypass()
                        removeCountdownOverlay()
                        handleWhatsAppForeground()
                    }
                }
            }
            mainHandler.post(countdownRunnable!!)
        } catch (e: Exception) {
            Log.e(TAG, "Failed to display countdown overlay: ${e.message}", e)
        }
    }

    private fun removeCountdownOverlay() {
        mainHandler.post {
            countdownRunnable?.let { mainHandler.removeCallbacks(it) }
            countdownRunnable = null
            try {
                if (countdownOverlayView != null) {
                    windowManager?.removeView(countdownOverlayView)
                    countdownOverlayView = null
                    Log.d(TAG, "Countdown overlay removed.")
                }
            } catch (e: Exception) {
                Log.e(TAG, "Error removing countdown overlay: ${e.message}", e)
            }
        }
    }

    private fun removeOverlay() {
        mainHandler.post {
            try {
                if (overlayView != null) {
                    windowManager?.removeView(overlayView)
                    overlayView = null
                    Log.d(TAG, "Blocking overlay removed.")
                }
            } catch (e: Exception) {
                Log.e(TAG, "Error removing overlay: ${e.message}", e)
            }
        }
    }

    override fun onInterrupt() {
        Log.w(TAG, "WhatsAppBlockerService interrupted.")
        removeOverlay()
    }

    override fun onDestroy() {
        super.onDestroy()
        removeOverlay()
    }
}

object PermissionHelper {
    fun isAccessibilityServiceEnabled(context: Context, service: Class<*>): Boolean {
        val expectedComponentName = "${context.packageName}/${service.name}"
        val enabledServices = Settings.Secure.getString(
            context.contentResolver,
            Settings.Secure.ENABLED_ACCESSIBILITY_SERVICES
        ) ?: return false
        return enabledServices.split(":").any { it.equals(expectedComponentName, ignoreCase = true) }
    }

    fun isNotificationListenerEnabled(context: Context): Boolean {
        val listeners = Settings.Secure.getString(
            context.contentResolver,
            "enabled_notification_listeners"
        ) ?: return false
        return listeners.split(":").any { it.contains(context.packageName) }
    }

    fun isOverlayPermissionGranted(context: Context): Boolean {
        return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
            Settings.canDrawOverlays(context)
        } else {
            true
        }
    }

    fun isBatteryOptimizationIgnored(context: Context): Boolean {
        return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
            val powerManager = context.getSystemService(Context.POWER_SERVICE) as? PowerManager
            powerManager?.isIgnoringBatteryOptimizations(context.packageName) ?: false
        } else {
            true
        }
    }
}
